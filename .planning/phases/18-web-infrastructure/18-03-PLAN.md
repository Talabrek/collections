Create WebAuthHandler class:

```java
package com.blockworlds.collections.web;

import at.favre.lib.crypto.bcrypt.BCrypt;
import io.javalin.Javalin;
import io.javalin.http.Context;
import io.javalin.http.UnauthorizedResponse;

public class WebAuthHandler {
    private final String passwordHash;

    public WebAuthHandler(String passwordHash) {
        this.passwordHash = passwordHash;
    }

    /**
     * Registers authentication middleware for API routes.
     * Static files are NOT protected (served before route matching).
     */
    public void register(Javalin app) {
        app.beforeMatched(ctx -> {
            // Only protect /api/ routes
            if (ctx.path().startsWith("/api/")) {
                validateAuth(ctx);
            }
        });
    }

    private void validateAuth(Context ctx) {
        if (!ctx.basicAuthCredentialsExist()) {
            challengeAuth(ctx);
            return;
        }

        var creds = ctx.basicAuthCredentials();

        // Username is ignored - single password auth
        BCrypt.Result result = BCrypt.verifyer()
            .verify(creds.getPassword().toCharArray(), passwordHash);

        if (!result.verified) {
            challengeAuth(ctx);
        }
    }

    private void challengeAuth(Context ctx) {
        ctx.header("WWW-Authenticate", "Basic realm=\"Collections Admin\", charset=\"UTF-8\"");
        ctx.status(401);
        throw new UnauthorizedResponse("Authentication required");
    }
}
```

Key points:
- Username field is ignored (any username works, only password matters)
- Uses BCrypt.verifyer() to compare against stored hash
- Throws UnauthorizedResponse which Javalin handles cleanly
- Only /api/ routes are protected, static files are public

```java
import com.blockworlds.collections.Collections;

public class StatusController {

    public StatusController(Collections plugin) {

            ctx.json(Map.of(

                "plugin", "Collections",
                "version", plugin.getPluginMeta().getVersion(),
                "collections", plugin.getCollectionManager().getCollectionCount(),
                "zones", plugin.getZoneManager().getZoneCount()
            ));
        });
    }
}
```

```java
private void configureJavalin(JavalinConfig config) {
    // Suppress banner
    config.showJavalinBanner = false;

    // Graceful shutdown timeout
    config.jetty.modifyServer(server -> {
        server.setStopTimeout(5000);
    });

    // Static files from classpath (inside JAR)
    config.staticFiles.add(staticFiles -> {
        staticFiles.hostedPath = "/";
        staticFiles.directory = "/web";
        staticFiles.location = io.javalin.http.staticfiles.Location.CLASSPATH;
    });
}
```

**src/main/resources/web/js/app.js:**

```javascript
// Fetch server status on page load
document.addEventListener('DOMContentLoaded', function() {
    const statusBox = document.getElementById('status');

    fetch('/api/status')
        .then(response => {
            if (response.status === 401) {
                statusBox.innerHTML = '<h3>Authentication Required</h3><p>Please log in to view server status.</p>';
                statusBox.classList.add('error');
                return null;
            }
            return response.json();
        })
        .then(data => {
            if (data) {
                statusBox.innerHTML = `
                    <h3>Server Status</h3>
                    <p><strong>Status:</strong> ${data.status}</p>
                    <p><strong>Version:</strong> ${data.version}</p>
                    <p><strong>Collections:</strong> ${data.collections}</p>
                    <p><strong>Zones:</strong> ${data.zones}</p>
                `;
                statusBox.classList.add('success');
            }
        })
        .catch(error => {
            statusBox.innerHTML = '<h3>Connection Error</h3><p>Could not connect to server.</p>';
            statusBox.classList.add('error');
        });
});
```
